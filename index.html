/** * MUCS Advanced Service Request Engine (صلاحيات جبارة للعميل) * هذه الدالة تتحكم في تنفيذ الطلب، اعتراضه، فحصه، وتنفيذ السكريبتات المرتبطة به ديناميكياً */ async function requestNewService(serviceName) { try { // 1. فحص حالة الطوارئ العامة للنظام (Global System Lock) const isSystemLocked = localStorage.getItem('mucs_global_emergency_lock') === 'true'; if (isSystemLocked) { alert("عذراً، منصة مُتقن تخضع لصيانة طارئة حالياً. نرجو المحاولة لاحقاً."); return; } // 2. فحص الـ Hooks المسبقة (Before-Action Hook) التي وضعها المالك أو المشرف if (typeof MUCS_ClientEngine !== 'undefined' && MUCS_ClientEngine.executeHookBeforeAction) { const canProceed = await MUCS_ClientEngine.executeHookBeforeAction('service_request', { service: serviceName }); if (!canProceed) { alert("عذراً، تم إلغاء أو إيقاف هذا الإجراء مؤقتاً بواسطة الإدارة المركزية."); return; } } // 3. جلب بيانات الجلسة الحالية لتوثيق الطلب قانونياً ومالياً const sessionData = JSON.parse(localStorage.getItem('motqan_client_session') || '{}'); if (!sessionData.phone) { alert("انتهت صلاحية الجلسة، يرجى تسجيل الدخول مجدداً."); window.location.replace('https://mutaqan.oxygen11.com'); return; } // 4. إنشاء هيكل الطلب الموحد وتحويله للسجل المركزي (Registry Payload) const orderPayload = { orderId: 'ORD-' + Math.floor(100000 + Math.random() * 900000), clientPhone: sessionData.phone, clientName: sessionData.name || 'عميل مُتقن', serviceRequested: serviceName, timestamp: new Date().toISOString(), status: 'pending_dispatch', sourceDomain: 'client.oxygen11.com' }; // حفظ الطلب محلياً وفي السجل المشترك لحل أي نزاعات قانونية مستقبلية const allOrders = JSON.parse(localStorage.getItem('motqan_global_orders_registry') || '[]'); allOrders.unshift(orderPayload); localStorage.setItem('motqan_global_orders_registry', JSON.stringify(allOrders)); // 5. التنفيذ الناجح وإشعار المستخدم alert(`تم استلام طلبك (${serviceName}) بنجاح برقم مرجعي [${orderPayload.orderId}]. جاري توجيه أقرب فني معتمد لموقعك!`); // 6. تشغيل الـ Hooks اللاحقة (After-Action Hook) في الخلفية دون تعطيل الواجهة if (typeof MUCS_ClientEngine !== 'undefined' && MUCS_ClientEngine.executeHookAfterAction) { MUCS_ClientEngine.executeHookAfterAction('service_request_success', orderPayload); } // الانتقال أوتوماتيكياً لصفحة تتبع الطلبات switchTab('orders'); } catch (error) { console.error("Critical error in requestNewService:", error.message); alert("حدث خطأ غير متوقع أثناء معالجة طلبك. تم تسجيل المشكلة للإدارة."); } }
